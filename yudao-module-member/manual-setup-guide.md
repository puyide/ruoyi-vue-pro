# NodeBB SSO 手动配置指南（无需 psql 命令）

## 问题：psql 命令不可用

如果你遇到 `psql 未安装` 的错误，可以使用以下替代方案。

---

## 🔧 方案一：使用 NodeBB 内置的数据库连接（推荐）⭐

NodeBB 本身连接了 PostgreSQL，我们可以通过它来执行 SQL。

### 步骤：

```bash
# 1. 创建建表脚本
cd "/Users/admin/ma jia/NodeBB"
cat > create-member-nodebb-table.js << 'EOF'
const nconf = require('nconf');
const path = require('path');

nconf.argv().env({ separator: '__' });
nconf.file({ file: path.join(__dirname, 'config.json') });
nconf.defaults({ base_dir: __dirname });

const db = require('./src/database');

async function createTable() {
  console.log('\n🔧 正在连接 NodeBB 数据库...');
  await db.init();
  
  const sql = `
-- 创建 member_nodebb_user 表
CREATE TABLE IF NOT EXISTS member_nodebb_user (
    id int8 GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    user_id int8 NOT NULL,
    nodebb_uid int4 DEFAULT NULL,
    username varchar(255) DEFAULT NULL,
    sync_status int2 DEFAULT 0,
    sync_error_msg varchar(500) DEFAULT NULL,
    creator varchar(64) DEFAULT '',
    create_time timestamp DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updater varchar(64) DEFAULT '',
    update_time timestamp DEFAULT CURRENT_TIMESTAMP NOT NULL,
    deleted int2 DEFAULT 0 NOT NULL,
    tenant_id int8 DEFAULT 0 NOT NULL,
    CONSTRAINT pk_member_nodebb_user PRIMARY KEY (id)
);

-- 创建索引
CREATE UNIQUE INDEX IF NOT EXISTS uk_user_id_deleted ON member_nodebb_user (user_id, deleted);
CREATE INDEX IF NOT EXISTS idx_nodebb_uid ON member_nodebb_user (nodebb_uid);

-- 添加注释
COMMENT ON TABLE member_nodebb_user IS '会员用户 - NodeBB 用户映射表';
COMMENT ON COLUMN member_nodebb_user.id IS '自增主键';
COMMENT ON COLUMN member_nodebb_user.user_id IS '会员用户 ID';
COMMENT ON COLUMN member_nodebb_user.nodebb_uid IS 'NodeBB 用户 UID';
COMMENT ON COLUMN member_nodebb_user.username IS 'NodeBB 用户名';
COMMENT ON COLUMN member_nodebb_user.sync_status IS '同步状态：1-成功，0-失败';
COMMENT ON COLUMN member_nodebb_user.sync_error_msg IS '同步失败原因';
  `;
  
  console.log('📝 执行建表 SQL...\n');
  await db.pool.query(sql);
  
  console.log('✅ member_nodebb_user 表创建成功！\n');
  
  // 验证表是否创建成功
  const result = await db.pool.query(`
    SELECT table_name, column_name, data_type 
    FROM information_schema.columns 
    WHERE table_name = 'member_nodebb_user' 
    ORDER BY ordinal_position;
  `);
  
  console.log('📊 表结构验证：');
  console.log('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  result.rows.forEach(row => {
    console.log(`  ${row.column_name.padEnd(20)} ${row.data_type}`);
  });
  console.log('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n');
  
  await db.close();
  process.exit(0);
}

createTable().catch(err => {
  console.error('\n❌ 建表失败：', err.message);
  console.error(err.stack);
  process.exit(1);
});
EOF

# 2. 执行脚本
node create-member-nodebb-table.js
```

这个脚本会：
- ✅ 使用 NodeBB 的数据库连接
- ✅ 创建 `member_nodebb_user` 表
- ✅ 创建索引和注释
- ✅ 验证表结构

## 🔧 方案三：安装 PostgreSQL 客户端

### macOS（使用 Homebrew）

```bash
# 只安装客户端工具（不安装服务器）
brew install libpq

# 添加到 PATH（临时）
export PATH="/opt/homebrew/opt/libpq/bin:$PATH"

# 或永久添加到 ~/.zshrc
echo 'export PATH="/opt/homebrew/opt/libpq/bin:$PATH"' >> ~/.zshrc
source ~/.zshrc

# 验证
psql --version
```

### macOS（使用 Postgres.app）

1. 下载：https://postgresapp.com/
2. 安装并启动
3. 添加到 PATH：
   ```bash
   echo 'export PATH="/Applications/Postgres.app/Contents/Versions/latest/bin:$PATH"' >> ~/.zshrc
   source ~/.zshrc
   ```
